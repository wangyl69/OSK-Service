cmake_minimum_required(VERSION 3.20)
project(HybridHaptic LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(HybridHaptic src/main.cpp src/runtime.cpp)
target_include_directories(HybridHaptic PRIVATE src)
target_compile_definitions(HybridHaptic PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
target_link_libraries(HybridHaptic PRIVATE hid setupapi advapi32 user32 wtsapi32)
add_executable(PanelProbe tools/panel_probe.cpp)
target_include_directories(PanelProbe PRIVATE src)
target_compile_definitions(PanelProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(PanelProbe PRIVATE hid setupapi user32)
add_executable(PanelTests tests/panel_tests.cpp)
target_include_directories(PanelTests PRIVATE tools)
target_compile_definitions(PanelTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(PanelTests PRIVATE user32)
if(MINGW)
  target_link_options(PanelProbe PRIVATE -static)
  target_link_options(PanelTests PRIVATE -static)
endif()
if(MSVC)
  target_compile_options(PanelProbe PRIVATE /W4 /utf-8)
else()
  target_compile_options(PanelProbe PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(HybridHapticTests tests/core_tests.cpp)
target_include_directories(HybridHapticTests PRIVATE src)
add_executable(HybridIoTests tests/io_tests.cpp)
target_include_directories(HybridIoTests PRIVATE src)
target_compile_definitions(HybridIoTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
set(FIRMWARE_PROTOCOL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../Athena_Haptic/Core/Inc")
if(EXISTS "${FIRMWARE_PROTOCOL_DIR}/app_i2chid_protocol.h")
  add_executable(HybridFirmwareRuntimeTests tests/firmware_runtime_tests.cpp)
  target_include_directories(HybridFirmwareRuntimeTests PRIVATE "${FIRMWARE_PROTOCOL_DIR}")
  add_executable(HybridFirmwareContractTests tests/firmware_contract_tests.cpp)
  target_include_directories(HybridFirmwareContractTests PRIVATE src "${FIRMWARE_PROTOCOL_DIR}")
  if(MINGW)
    target_link_options(HybridFirmwareRuntimeTests PRIVATE -static)
    target_link_options(HybridFirmwareContractTests PRIVATE -static)
  endif()
endif()
if(MINGW)
  target_link_options(HybridHaptic PRIVATE -municode -static)
  target_link_options(HybridHapticTests PRIVATE -static)
  target_link_options(HybridIoTests PRIVATE -static)
endif()
if(MSVC)
  target_compile_options(HybridHaptic PRIVATE /W4 /utf-8)
  target_compile_options(HybridHapticTests PRIVATE /W4 /utf-8)
else()
  target_compile_options(HybridHaptic PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(HybridHapticTests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME core COMMAND HybridHapticTests)
add_test(NAME windows_io COMMAND HybridIoTests)
add_test(NAME panel_raw_input COMMAND PanelTests)
if(TARGET HybridFirmwareContractTests)
  add_test(NAME firmware_contract COMMAND HybridFirmwareContractTests)
  add_test(NAME firmware_idle_handshake COMMAND HybridFirmwareRuntimeTests)
endif()
